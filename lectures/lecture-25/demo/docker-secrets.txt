# Secrets leak demo, lecture 25
# Docker version 29.4.0, build 9d7ad9f

## 1. A .env file in the build context
$ cat .env
API_KEY=fake-key-123

$ docker build -t datasci350-project-leak .   # all layers cached, 0.9s total

## 2. The key is inside the image
$ docker run --rm datasci350-project-leak cat .env
API_KEY=fake-key-123

$ docker run --rm datasci350-project-leak ls -la
total 40

-rw-r--r-- 1 root root  536 Sep 13 18:25 .dockerignore
-rw-r--r-- 1 root root   21 Sep 13 18:25 .env
-rw-r--r-- 1 root root  451 Sep 13 18:25 .gitignore
-rw-r--r-- 1 root root 5467 Sep 13 18:25 Dockerfile

## 3. The ENV route leaks too
$ cp Dockerfile Dockerfile.env
$ echo 'ENV API_KEY=fake-key-123' >> Dockerfile.env
$ docker build -f Dockerfile.env -t datasci350-project-leak-env .
1 warning found (use docker --debug to expand):
 - SecretsUsedInArgOrEnv: Do not use ARG or ENV instructions for sensitive data (ENV "API_KEY") (line 113)

$ docker history datasci350-project-leak-env | head -3
IMAGE          CREATED         CREATED BY                                      SIZE      COMMENT
ef57862caf53   9 seconds ago   ENV API_KEY=fake-key-123                        0B        buildkit.dockerfile.v0
<missing>      9 seconds ago   CMD ["bash" "-c" "quarto render report.qmd &…   0B        buildkit.dockerfile.v0

# Anyone who pulls the image can read it back:
$ docker run --rm datasci350-project-leak-env printenv API_KEY
fake-key-123

## 4. The fix: keep .env out of the build context
$ echo '.env' >> .dockerignore
$ docker build -t datasci350-project-leak .   # 0.6s, layers cached

$ docker run --rm datasci350-project-leak ls -la .env
ls: cannot access '.env': No such file or directory

$ docker run --rm datasci350-project-leak ls -a
.
..
.dockerignore
.gitignore
Dockerfile
Dockerfile.env
README.md
data
report.qmd
requirements.txt
scripts

## 5. Pass the secret at run time instead
$ docker run --rm -e API_KEY=fake-key-123 datasci350-project bash -c 'echo $API_KEY'
fake-key-123

$ docker run --rm --env-file .env datasci350-project bash -c 'echo $API_KEY'
fake-key-123

# The key is in the running container, not in the image:
$ docker history datasci350-project | grep -c API_KEY
0
